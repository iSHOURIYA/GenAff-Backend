# === LEGACY MANUAL STEPS (use deploy.sh instead) ===
# Manual steps are kept below for reference only.
# The deploy.sh script automates everything.

# 1. Install Node.js 20
curl -fsSL https://deb.nodesource.com/setup_20.x | sudo -E bash -
sudo apt-get install -y nodejs

# 2. Install PostgreSQL
sudo apt install postgresql postgresql-contrib -y

# 3. Install PM2 + Nginx
sudo npm install -g pm2
sudo apt install nginx -y

# 4. Create DB
sudo -u postgres psql -c "CREATE USER genaff_user WITH PASSWORD 'your_strong_password';"
sudo -u postgres psql -c "CREATE DATABASE genaff OWNER genaff_user;"

# 5. Clone and setup
git clone https://github.com/yourusername/genaff-backend.git
cd genaff-backend
npm install
cp .env.example .env
nano .env   # fill in LIVE values — NODE_ENV=production, rzp_live_ keys, real DB URL
npm run build        # prisma generate
npm run db:migrate   # prisma migrate deploy
pm2 start src/server.js --name genaff-backend
pm2 save && pm2 startup

# 6. Nginx config
cat | sudo tee /etc/nginx/sites-available/genaff-api <<'EOF'
server {
    listen 80;
    server_name YOUR_API_DOMAIN_HERE;  # e.g., genaff-api.shouriya.tech
    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 120s;
    }
}
EOF
sudo ln -s /etc/nginx/sites-available/genaff-api /etc/nginx/sites-enabled/
sudo nginx -t && sudo systemctl reload nginx

# 7. SSL
sudo apt install certbot python3-certbot-nginx -y
sudo certbot --nginx -d YOUR_API_DOMAIN_HERE
