# Install PM2 + Nginx
sudo npm install -g pm2
sudo apt install nginx -y

# Create DB
sudo -u postgres psql -c "CREATE USER genaff_user WITH PASSWORD 'your_strong_password';"
sudo -u postgres psql -c "CREATE DATABASE genaff OWNER genaff_user;"


git clone https://github.com/yourusername/genaff-backend.git
cd genaff-backend
npm install
cp .env.example .env
nano .env   # fill in LIVE values — NODE_ENV=production, rzp_live_ keys, real DB URL
npm run build        # prisma generate
npm run db:migrate   # prisma migrate deploy
pm2 start src/server.js --name genaff-backend
pm2 save && pm2 startup

sudo nano /etc/nginx/sites-available/genaff-api

server {
    listen 80;
    server_name genaff-api.shauryacodes.xyz;
    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 120s;
    }
}

sudo ln -s /etc/nginx/sites-available/genaff-api /etc/nginx/sites-enabled/
sudo nginx -t && sudo systemctl reload nginx
